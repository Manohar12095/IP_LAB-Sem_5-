----- order_process.php -----

<?php
$servername = "localhost";
$username = "root";
$password = "mysql";
$dbname = "shop";
$conn = new mysqli($servername, $username, $password, $dbname);
if ($conn->connect_error) {
  die("Connection failed: " . $conn->connect_error);
}

$message = "";
if ($_SERVER["REQUEST_METHOD"] == "POST") {
  $customer_name = $conn->real_escape_string($_POST['customer_name']);
  $product_name = $conn->real_escape_string($_POST['product_name']);
  $quantity = (int)$_POST['quantity'];
  $price = (float)$_POST['price'];
  $stmt = $conn->prepare("INSERT INTO orders (customer_name, product_name, quantity, price)
VALUES (?, ?, ?, ?)");
  $stmt->bind_param("ssid", $customer_name, $product_name, $quantity, $price);

  if ($stmt->execute()) {
    $message = "Order successfully recorded!";
  } else {
    $message = "Error: " . $stmt->error;
  }
  $stmt->close();
}
$sql = "SELECT id, customer_name, product_name, quantity, price, order_date FROM orders ORDER BY
order_date DESC";
$result = $conn->query($sql);
$conn->close();
?>
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <title>All Orders</title>
  <style>
    body { font-family: Arial, sans-serif; margin: 40px; }
    .message { margin-bottom: 20px; color: green; }
    table { border-collapse: collapse; width: 100%; max-width: 900px; }
    th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
    th { background-color: #f2f2f2; }
    a { display: inline-block; margin-top: 20px; }
  </style>
</head>
<body>

<h2>All Orders</h2>

<?php if ($message): ?>
  <div class="message"><?= htmlspecialchars($message) ?></div>
<?php endif; ?>

<?php if ($result && $result->num_rows > 0): ?>
  <table>
    <thead>
      <tr>
        <th>ID</th>
        <th>Customer Name</th>
        <th>Product Name</th>
        <th>Quantity</th>
        <th>Price (per item)</th>
        <th>Total Price</th>
        <th>Order Date</th>
      </tr>
    </thead>
    <tbody>
      <?php while($row = $result->fetch_assoc()): ?>
        <tr>
          <td><?= htmlspecialchars($row['id']) ?></td>
          <td><?= htmlspecialchars($row['customer_name']) ?></td>
          <td><?= htmlspecialchars($row['product_name']) ?></td>
          <td><?= htmlspecialchars($row['quantity']) ?></td>
          <td>$<?= number_format($row['price'], 2) ?></td>
          <td>$<?= number_format($row['price'] * $row['quantity'], 2) ?></td>
          <td><?= htmlspecialchars($row['order_date']) ?></td>
        </tr>
      <?php endwhile; ?>
    </tbody>
  </table>
<?php else: ?>
  <p>No orders found.</p>
<?php endif; ?>
<a href="order_form.html">Place another order</a>
</body>
</html>
